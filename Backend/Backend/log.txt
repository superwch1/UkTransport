// LineName is intentionally NOT filtered. The live feed's PublishedLineName can differ from the timetable's LineName for the same physical route
            // The journeys are resolved first and their calling points fetched afterwards, because joining the two in one
            // query repeats every timetable column once per calling point and forces the day filter's subqueries over the joined rows.
            var candidates = await _context.BusTimetables
                .Where(x => tripScheduleKeys.Contains(x.TripScheduleKey) &&
                            x.StartDate <= _timeService.UkNowDateOnly && x.EndDate >= _timeService.UkNowDateOnly)
                .ApplyDayFilter(_timeService.UkNowDateTime, false)
                .AsNoTracking()
                .Select(x => new { x.Id, x.TripScheduleKey, x.StartDate })
                .ToListAsync();

            if (candidates.Count == 0)
                return new Dictionary<string, IReadOnlyList<BusCallingPoint>>();

            // If several journeys still match, prefer the most recently-started schedule so repeated taps deterministically resolve to the current timetable version.
            Dictionary<string, string> timetableIdByKey = candidates
                .GroupBy(x => x.TripScheduleKey)
                .ToDictionary(x => x.Key, x => x.OrderByDescending(y => y.StartDate).First().Id);

            string[] timetableIds = [.. timetableIdByKey.Values];

            // Ordered here so callers can rely on the sequence order rather than on however the rows happen to come back.
            List<BusCallingPoint> callingPoints = await _context.BusCallingPoints
                .Where(x => timetableIds.Contains(x.BusTimetableId))
                .AsNoTracking()
                .OrderBy(x => x.Sequence)
                .ToListAsync();

            var callingPointsByTimetableId = new Dictionary<string, List<BusCallingPoint>>(timetableIds.Length);
            foreach (BusCallingPoint callingPoint in callingPoints)
            {
                if (!callingPointsByTimetableId.TryGetValue(callingPoint.BusTimetableId, out List<BusCallingPoint>? journeyCallingPoints))
                {
                    journeyCallingPoints = [];
                    callingPointsByTimetableId[callingPoint.BusTimetableId] = journeyCallingPoints;
                }

                journeyCallingPoints.Add(callingPoint);
            }

            // A journey with no calling points is left out entirely, so callers never receive an empty route to index into.
            var routeByKey = new Dictionary<string, IReadOnlyList<BusCallingPoint>>(timetableIdByKey.Count);
            foreach ((string tripScheduleKey, string timetableId) in timetableIdByKey)
            {
                if (callingPointsByTimetableId.TryGetValue(timetableId, out List<BusCallingPoint>? journeyCallingPoints))
                {
                    routeByKey[tripScheduleKey] = journeyCallingPoints;
                }
            }

            return routeByKey;
